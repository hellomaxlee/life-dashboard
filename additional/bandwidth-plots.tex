\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{fillbetween}
\definecolor{lampred}{RGB}{255,0,0}
\definecolor{lampgold}{RGB}{255,200,40}
\definecolor{lampgreen}{RGB}{60,220,90}
\pgfplotsset{
  colormap={lamp}{color(0)=(lampred) color(1)=(lampred) color(2)=(lampred)
    color(3)=(lampgold) color(4)=(lampgreen) color(5)=(lampgreen) color(6)=(lampgreen)},
}
\pgfmathdeclarefunction{zs}{1}{\pgfmathparse{%
  (#1<=4.5)*(-3)
  + and(#1>4.5,#1<=5.5)*(-3+(#1-4.5))
  + and(#1>5.5,#1<=6)*(-2+4*(#1-5.5))
  + and(#1>6.5,#1<=7)*(4*(#1-6.5))
  + and(#1>7,#1<=8.5)*2
  + and(#1>8.5,#1<=9)*(2-4*(#1-8.5))
  + and(#1>9.5,#1<=10)*(-4*(#1-9.5))
  + and(#1>10,#1<=11)*(-2-(#1-10))
  + (#1>11)*(-3)}}
\pgfmathdeclarefunction{zh}{1}{\pgfmathparse{max(-3,min(3,-(#1)/2))}}
\pgfmathdeclarefunction{bw}{2}{\pgfmathparse{50+25/3*(zs(#1)+zh(#2))}}
